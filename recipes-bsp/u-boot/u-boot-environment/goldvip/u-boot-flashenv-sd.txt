// The following macros will be automagically replaced when building the binary environment:
//       $KERNEL_DEVICETREE         Basename of KERNEL_DEVICETREE

#include "goldvip-distro-env-macros.txt"

fdt_file=$KERNEL_DEVICETREE

#if GOLDVIP_SKIP_SERDES1_CONFIG
#define SERDES0_CONFIG serdes0:mode=pcie,clock=ext;pcie0:mode=rc
#define SERDES1_CONFIG serdes1:skip=1

hwconfig=SERDES0_CONFIG;SERDES1_CONFIG
#endif /* GOLDVIP_SKIP_SERDES1_CONFIG */

#if GOLDVIP_GATEWAY_ENABLED
#define GATEWAY_DEPLOYMENT_CMD write_gateway_app

// Macro for flashing the GoldVIP real-time gateway application
GATEWAY_DEPLOYMENT_CMD=echo "Flashing the GoldVIP real-time gateway application..."; \
  setenv image goldvip-gateway.bin && \
  setenv flash_addr 0x00400000 && \
  run write_flash_image
#else /* GOLDVIP_GATEWAY_ENABLED */
#define GATEWAY_DEPLOYMENT_CMD
#endif /* GOLDVIP_GATEWAY_ENABLED */

#if GOLDVIP_BOOTLOADER_ENABLED
#define BOOTLOADER_DEPLOYMENT_CMD write_bootloader

// Macro for flashing the GoldVIP real-time bootloader
BOOTLOADER_DEPLOYMENT_CMD=echo "Flashing the GoldVIP real-time bootloader..."; \
  setenv image boot-loader && \
  setenv flash_addr 0x00000000 && \
  run write_flash_image
#else /* GOLDVIP_BOOTLOADER_ENABLED */
#define BOOTLOADER_DEPLOYMENT_CMD
#endif /* GOLDVIP_BOOTLOADER_ENABLED */

#if (GOLDVIP_BOOTLOADER_ENABLED && GOLDVIP_BOOTCONFIG_ENABLED)
#define BOOTCONFIG_DEPLOYMENT_CMD write_bootconfig

// Macro for flashing the boot configuration for the GoldVIP real-time bootloader
BOOTCONFIG_DEPLOYMENT_CMD=echo "Flashing the configuration for the GoldVip real-time bootloader..."; \
  setenv image Bootloader_Configuration.bin && \
  setenv flash_addr 0x001F0000 && \
  run write_flash_image
#else /* (GOLDVIP_BOOTLOADER_ENABLED && GOLDVIP_BOOTCONFIG_ENABLED) */
#define BOOTCONFIG_DEPLOYMENT_CMD
#endif /*(GOLDVIP_BOOTLOADER_ENABLED && GOLDVIP_BOOTCONFIG_ENABLED) */

#define ATF_DEPLOYMENT_CMD write_atf
// MACRO for flashing the ATF binary
ATF_DEPLOYMENT_CMD=echo "Flashing the ARM Trusted Firmware binary..."; \
  setenv image fip.s32-sdcard && \
  setenv flash_addr 0x00200000 && \
  run write_flash_image

write_flash_image=sf probe 6:0 && \
  run loadimage && \
  sf erase ${flash_addr} +${filesize} && \
  sf write ${loadaddr} ${flash_addr} ${filesize}; \
  env default image flash_addr

write_goldvip_images=echo 'Flashing the GoldVIP images...'; run BOOTLOADER_DEPLOYMENT_CMD BOOTCONFIG_DEPLOYMENT_CMD GATEWAY_DEPLOYMENT_CMD ATF_DEPLOYMENT_CMD
